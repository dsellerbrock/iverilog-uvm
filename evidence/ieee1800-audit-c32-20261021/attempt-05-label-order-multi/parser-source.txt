    41	static vpiHandle find_scope(vpiHandle scope, const char*name)
    42	{
    43	      vpiHandle idx = vpi_iterate(vpiModule, scope);
    44		/* If this scope has no modules then it can't have the one we
    45		 * are looking for so just return 0. */
    46	      if (idx == 0) return 0;
    47	
    48	      vpiHandle cur;
    49	      while ( (cur = vpi_scan(idx)) ) {
    50	
    51		    if ( strcmp(name, vpi_get_str(vpiName,cur)) == 0) {
    52			  vpi_free_object(idx);
    53			  return cur;
    54		    }
    55	      }
    56	
    57	      return 0;
    58	}

    69	 * These functions are called by the SDF parser during parsing to

    73	void sdf_select_instance(const char*celltype, const char*cellinst, const int sdf_lineno)
    74	{
    75	      char buffer[128];
    76	
    77		/* Test for wildcard character */
    78	      if (cellinst == NULL) {
    79		    if (sdf_flag_warning) {
    80			  vpi_printf("SDF WARNING: %s:%d: sorry: "
    81			             "Wildcard cell instance specification (*) currently not supported.\n",
    82			             sdf_fname, sdf_lineno);
    83		    }
    84		    sdf_cur_cell = 0;
    85		    return;
    86	      }
    87	
    88		/* First follow the hierarchical parts of the cellinst name to
    89		   get to the cell that I'm looking for. */
    90	      vpiHandle scope = sdf_scope;
    91	      const char*src = cellinst;
    92	      const char*dp;
    93	      while ( (dp=strchr(src, '.')) ) {
    94		    unsigned len = dp - src;
    95		    assert(dp >= src);
    96		    assert(len < sizeof buffer);
    97		    strncpy(buffer, src, len);
    98		    buffer[len] = 0;
    99	
   100		    vpiHandle tmp_scope = find_scope(scope, buffer);
   101		    if (tmp_scope == 0) {
   102			  vpi_printf("SDF ERROR: %s:%d: Cannot find %s in scope %s.\n",
   103			             sdf_fname, sdf_lineno, buffer, vpi_get_str(vpiFullName, scope));
   104			  break;
   105		    }
   106		    assert(tmp_scope);
   107		    scope = tmp_scope;
